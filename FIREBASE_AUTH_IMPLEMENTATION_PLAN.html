Why Authentication is Critical

Problems with Current Device-Based Approach

❌ No Cross-Device Sync - User subscribes on iPhone → Settings lost on iPad - User gets new phone → Must subscribe again - Loses saved unit preferences

❌ RevenueCat Incompatibility - RevenueCat designed for user accounts - Device ID changes = lost subscription - Cannot restore purchases reliably

❌ Poor User Experience - Cannot backup settings - Cannot use app on multiple devices - Reinstalling app = start over

❌ Business Risk - Users may request refunds for lost subscriptions - Cannot provide customer support (no way to identify users) - Cannot track user lifetime value

Benefits of Authentication

✅ Subscription Continuity - User buys on iPhone → Works on iPad automatically - New phone? Sign in and subscription follows - RevenueCat syncs across devices

✅ Better UX - Save preferences in cloud - Seamless multi-device experience - Account recovery

✅ Business Benefits - Customer support possible (look up by email) - Track user lifetime value - Send important emails (receipt, renewal reminder) - Lower churn (easier to use)

Proposed Solution

Authentication Methods

1. Email/Password (Primary) - Traditional signup/signin - Password reset via email - Most universal option

2. Google Sign-In (Secondary) - One-tap sign-in - No password to remember - Apple App Store requirement: If you have Google, must also have Apple Sign-In

3. Anonymous → Email Upgrade - Start anonymous (no friction) - Later prompt to “Secure Your Account” - Link anonymous account to email

import 'dart:developer';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:google_sign_in/google_sign_in.dart';

class AuthService {
  final FirebaseAuth _auth = FirebaseAuth.instance;
  final GoogleSignIn _googleSignIn = GoogleSignIn(
    scopes: ['email', 'profile'],
  );

  // ============== USER STATE ==============

  /// Get current user
  User? get currentUser => _auth.currentUser;

  /// Get current user ID
  String? get currentUserId => _auth.currentUser?.uid;

  /// Check if user is signed in
  bool get isSignedIn => _auth.currentUser != null;

  /// Check if user is anonymous
  bool get isAnonymous => _auth.currentUser?.isAnonymous ?? false;

  /// Stream of auth state changes
  Stream<User?> get authStateChanges => _auth.authStateChanges();

  // ============== ANONYMOUS AUTH ==============

  /// Sign in anonymously (for guest users)
  Future<UserCredential> signInAnonymously() async {
    try {
      log('Signing in anonymously...');
      final userCredential = await _auth.signInAnonymously();
      log('Anonymous sign-in successful: ${userCredential.user?.uid}');
      return userCredential;
    } catch (e) {
      log('Anonymous sign-in error: $e');
      rethrow;
    }
  }

  // ============== EMAIL/PASSWORD AUTH ==============

  /// Sign up with email and password
  Future<UserCredential> signUpWithEmail({
    required String email,
    required String password,
    String? displayName,
  }) async {
    try {
      log('Creating account for: $email');

      // Create user
      final userCredential = await _auth.createUserWithEmailAndPassword(
        email: email.trim(),
        password: password,
      );

      // Update display name
      if (displayName != null && displayName.isNotEmpty) {
        await userCredential.user?.updateDisplayName(displayName.trim());
      }

      // Send email verification
      await userCredential.user?.sendEmailVerification();
      log('Verification email sent to: $email');

      return userCredential;
    } on FirebaseAuthException catch (e) {
      log('Sign up error: ${e.code} - ${e.message}');
      throw _handleAuthException(e);
    } catch (e) {
      log('Unexpected sign up error: $e');
      rethrow;
    }
  }

  /// Sign in with email and password
  Future<UserCredential> signInWithEmail({
    required String email,
    required String password,
  }) async {
    try {
      log('Signing in: $email');

      final userCredential = await _auth.signInWithEmailAndPassword(
        email: email.trim(),
        password: password,
      );

      log('Sign-in successful: ${userCredential.user?.uid}');
      return userCredential;
    } on FirebaseAuthException catch (e) {
      log('Sign in error: ${e.code} - ${e.message}');
      throw _handleAuthException(e);
    } catch (e) {
      log('Unexpected sign in error: $e');
      rethrow;
    }
  }

  // ============== GOOGLE SIGN-IN ==============

  /// Sign in with Google
  Future<UserCredential> signInWithGoogle() async {
    try {
      log('Starting Google Sign-In...');

      // Trigger Google Sign-In flow
      final GoogleSignInAccount? googleUser = await _googleSignIn.signIn();

      if (googleUser == null) {
        throw Exception('Google Sign-In cancelled by user');
      }

      log('Google user selected: ${googleUser.email}');

      // Obtain auth details
      final GoogleSignInAuthentication googleAuth = await googleUser.authentication;

      // Create credential
      final credential = GoogleAuthProvider.credential(
        accessToken: googleAuth.accessToken,
        idToken: googleAuth.idToken,
      );

      // Sign in to Firebase
      final userCredential = await _auth.signInWithCredential(credential);

      log('Google Sign-In successful: ${userCredential.user?.uid}');
      return userCredential;
    } on FirebaseAuthException catch (e) {
      log('Google Sign-In error: ${e.code} - ${e.message}');
      throw _handleAuthException(e);
    } catch (e) {
      log('Unexpected Google Sign-In error: $e');
      rethrow;
    }
  }

  // ============== PASSWORD RESET ==============

  /// Send password reset email
  Future<void> sendPasswordResetEmail(String email) async {
    try {
      log('Sending password reset email to: $email');

      await _auth.sendPasswordResetEmail(email: email.trim());

      log('Password reset email sent successfully');
    } on FirebaseAuthException catch (e) {
      log('Password reset error: ${e.code} - ${e.message}');
      throw _handleAuthException(e);
    } catch (e) {
      log('Unexpected password reset error: $e');
      rethrow;
    }
  }

  /// Confirm password reset with code
  Future<void> confirmPasswordReset({
    required String code,
    required String newPassword,
  }) async {
    try {
      log('Confirming password reset...');

      await _auth.confirmPasswordReset(
        code: code,
        newPassword: newPassword,
      );

      log('Password reset successful');
    } on FirebaseAuthException catch (e) {
      log('Password reset confirmation error: ${e.code} - ${e.message}');
      throw _handleAuthException(e);
    } catch (e) {
      log('Unexpected password reset confirmation error: $e');
      rethrow;
    }
  }

  // ============== ACCOUNT MANAGEMENT ==============

  /// Update user email
  Future<void> updateEmail(String newEmail) async {
    try {
      await currentUser?.verifyBeforeUpdateEmail(newEmail.trim());
      log('Email update verification sent');
    } on FirebaseAuthException catch (e) {
      throw _handleAuthException(e);
    }
  }

  /// Update user password
  Future<void> updatePassword(String newPassword) async {
    try {
      await currentUser?.updatePassword(newPassword);
      log('Password updated successfully');
    } on FirebaseAuthException catch (e) {
      throw _handleAuthException(e);
    }
  }

  /// Send email verification
  Future<void> sendEmailVerification() async {
    try {
      await currentUser?.sendEmailVerification();
      log('Verification email sent');
    } on FirebaseAuthException catch (e) {
      throw _handleAuthException(e);
    }
  }

  /// Reload user data
  Future<void> reloadUser() async {
    await currentUser?.reload();
  }

  // ============== ACCOUNT LINKING ==============

  /// Link anonymous account to email/password
  Future<UserCredential> linkAnonymousToEmail({
    required String email,
    required String password,
  }) async {
    try {
      if (!isAnonymous) {
        throw Exception('User is not anonymous');
      }

      log('Linking anonymous account to email: $email');

      final credential = EmailAuthProvider.credential(
        email: email.trim(),
        password: password,
      );

      final userCredential = await currentUser!.linkWithCredential(credential);

      // Send verification email
      await userCredential.user?.sendEmailVerification();

      log('Anonymous account linked to email successfully');
      return userCredential;
    } on FirebaseAuthException catch (e) {
      log('Account linking error: ${e.code} - ${e.message}');
      throw _handleAuthException(e);
    } catch (e) {
      log('Unexpected account linking error: $e');
      rethrow;
    }
  }

  /// Link anonymous account to Google
  Future<UserCredential> linkAnonymousToGoogle() async {
    try {
      if (!isAnonymous) {
        throw Exception('User is not anonymous');
      }

      log('Linking anonymous account to Google...');

      // Trigger Google Sign-In flow
      final GoogleSignInAccount? googleUser = await _googleSignIn.signIn();

      if (googleUser == null) {
        throw Exception('Google Sign-In cancelled by user');
      }

      // Obtain auth details
      final GoogleSignInAuthentication googleAuth = await googleUser.authentication;

      // Create credential
      final credential = GoogleAuthProvider.credential(
        accessToken: googleAuth.accessToken,
        idToken: googleAuth.idToken,
      );

      // Link accounts
      final userCredential = await currentUser!.linkWithCredential(credential);

      log('Anonymous account linked to Google successfully');
      return userCredential;
    } on FirebaseAuthException catch (e) {
      log('Google account linking error: ${e.code} - ${e.message}');
      throw _handleAuthException(e);
    } catch (e) {
      log('Unexpected Google account linking error: $e');
      rethrow;
    }
  }

  // ============== SIGN OUT ==============

  /// Sign out
  Future<void> signOut() async {
    try {
      log('Signing out...');

      // Sign out from Google if signed in
      if (await _googleSignIn.isSignedIn()) {
        await _googleSignIn.signOut();
      }

      // Sign out from Firebase
      await _auth.signOut();

      log('Sign-out successful');
    } catch (e) {
      log('Sign-out error: $e');
      rethrow;
    }
  }

  /// Delete account
  Future<void> deleteAccount() async {
    try {
      log('Deleting account...');

      await currentUser?.delete();

      log('Account deleted successfully');
    } on FirebaseAuthException catch (e) {
      log('Account deletion error: ${e.code} - ${e.message}');
      throw _handleAuthException(e);
    } catch (e) {
      log('Unexpected account deletion error: $e');
      rethrow;
    }
  }

  // ============== RE-AUTHENTICATION ==============

  /// Re-authenticate with email/password (required before sensitive operations)
  Future<UserCredential> reauthenticateWithEmail({
    required String email,
    required String password,
  }) async {
    try {
      final credential = EmailAuthProvider.credential(
        email: email.trim(),
        password: password,
      );

      return await currentUser!.reauthenticateWithCredential(credential);
    } on FirebaseAuthException catch (e) {
      throw _handleAuthException(e);
    }
  }

  /// Re-authenticate with Google
  Future<UserCredential> reauthenticateWithGoogle() async {
    try {
      final GoogleSignInAccount? googleUser = await _googleSignIn.signIn();

      if (googleUser == null) {
        throw Exception('Google Sign-In cancelled');
      }

      final GoogleSignInAuthentication googleAuth = await googleUser.authentication;

      final credential = GoogleAuthProvider.credential(
        accessToken: googleAuth.accessToken,
        idToken: googleAuth.idToken,
      );

      return await currentUser!.reauthenticateWithCredential(credential);
    } on FirebaseAuthException catch (e) {
      throw _handleAuthException(e);
    }
  }

  // ============== ERROR HANDLING ==============

  /// Convert Firebase auth exceptions to user-friendly messages
  String _handleAuthException(FirebaseAuthException e) {
    switch (e.code) {
      // Sign up errors
      case 'email-already-in-use':
        return 'This email is already registered. Please sign in instead.';
      case 'invalid-email':
        return 'Invalid email address format.';
      case 'operation-not-allowed':
        return 'Email/password sign-in is not enabled. Contact support.';
      case 'weak-password':
        return 'Password is too weak. Use at least 6 characters.';

      // Sign in errors
      case 'user-disabled':
        return 'This account has been disabled. Contact support.';
      case 'user-not-found':
        return 'No account found with this email.';
      case 'wrong-password':
        return 'Incorrect password. Try again or reset your password.';
      case 'invalid-credential':
        return 'Invalid email or password.';

      // Password reset errors
      case 'expired-action-code':
        return 'Password reset link has expired. Request a new one.';
      case 'invalid-action-code':
        return 'Invalid or already used password reset link.';

      // Account linking errors
      case 'credential-already-in-use':
        return 'This account is already linked to another user.';
      case 'provider-already-linked':
        return 'This sign-in method is already linked to your account.';

      // General errors
      case 'network-request-failed':
        return 'Network error. Check your internet connection.';
      case 'too-many-requests':
        return 'Too many attempts. Please try again later.';
      case 'requires-recent-login':
        return 'Please sign in again to complete this action.';

      default:
        log('Unhandled auth error: ${e.code}');
        return 'Authentication error: ${e.message ?? "Unknown error"}';
    }
  }
}

  final _passwordController = TextEditingController();

    _emailController.dispose();
    _passwordController.dispose();
    super.dispose();

    if (!_formKey.currentState!.validate()) return;

    setState(() {
      _isLoading = true;
      _errorMessage = null;

      await _authService.signInWithEmail(
        email: _emailController.text.trim(),
        password: _passwordController.text,

      if (mounted) {
        context.goNamed(AppRoute.tabs.name);
      }
    } catch (e) {
      setState(() {
        _errorMessage = e.toString();

      if (mounted) {
        setState(() {
          _isLoading = false;
        });
      }

Migration Strategy

Option 1: Hard Migration (Recommended)

Approach: Force all users to create accounts

Steps: 1. Deploy new version with auth 2. Users open app → see sign-in screen 3. Can sign up or continue as guest 4. Guest mode → prompt to “Secure Account” after 7 days

Pros: - Clean migration - Better data integrity - Easier to support

Cons: - Some friction for existing users

Option 2: Soft Migration

Approach: Migrate device IDs to user IDs automatically

Steps: 1. When user signs up/in, look for existing device_id in Firestore 2. If found, merge data into new user document 3. Delete old device-based document

Code:

Future<void> migrateDeviceDataToUser(String userId) async {
  final deviceId = await SharedPrefs.instance.deviceId;

  // Find old device-based document
  final oldDocs = await usersCollection
      .where('device_id', isEqualTo: deviceId)
      .get();

  if (oldDocs.docs.isEmpty) return;

  // Merge data
  final oldData = oldDocs.docs.first.data();
  await usersCollection.doc(userId).set({
    'stations': oldData['stations'] ?? [],
    'email': FirebaseAuth.instance.currentUser?.email,
    'migrated_from_device': deviceId,
    'created_at': FieldValue.serverTimestamp(),
  });

  // Delete old document
  await oldDocs.docs.first.reference.delete();

  log('Migrated device data to user account');
}

service cloud.firestore {
  match /databases/{database}/documents {
    // Users can only read/write their own data
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Incidents are read-only for all authenticated users
    match /incidents/{incidentId} {
      allow read: if request.auth != null;
      allow write: if false; // Only backend can write
    }
  }

String? validatePassword(String? value) {
  if (value == null || value.isEmpty) {
    return 'Please enter a password';
  }
  if (value.length < 8) {
    return 'Password must be at least 8 characters';
  }
  if (!value.contains(RegExp(r'[A-Z]'))) {
    return 'Password must contain at least one uppercase letter';
  }
  if (!value.contains(RegExp(r'[0-9]'))) {
    return 'Password must contain at least one number';
  }
  return null;
}
